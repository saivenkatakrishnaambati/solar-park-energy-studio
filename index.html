// server.js - SolarPark Predictive Microgrid Backend
const express = require('express');
const cors = require('cors');
const app = express();
app.use(cors(), express.json());

// Default Historical Telemetry Store (30-Day Rolling Window)
const historicalLogs = [
  { date: '2026-11-06', facility: 'campus', solarKWh: 4720, loadKWh: 4410, peakShavedKW: 195 },
  { date: '2026-11-07', facility: 'campus', solarKWh: 4910, loadKWh: 4520, peakShavedKW: 210 },
  { date: '2026-11-08', facility: 'mall',   solarKWh: 5150, loadKWh: 6120, peakShavedKW: 240 }
];

// Predictive Solar & Load Dispatch Endpoint
app.post('/api/v1/predict-dispatch', (req, res) => {
  const { facility, hour, solarCapacityKW, irradiance, evCount, mode } = req.body;
  // Gaussian Solar Irradiance Curve (Peaks at 12:30 PM)
  const solarFactor = Math.exp(-Math.pow((hour - 12.5) / 3.1, 2));
  const predictedSolarKW = Math.round(solarCapacityKW * solarFactor * irradiance);

  // Facility Demand Profile + Afternoon HVAC Surge
  const baseLoad = facility === 'mall' ? 520 : facility === 'techpark' ? 460 : 360;
  const hvacSpike = Math.exp(-Math.pow((hour - 15.0) / 0.85, 2)) * 260;
  const predictedFacilityKW = Math.round(baseLoad + hvacSpike);

  // AI Departure-Driven Throttling Algorithm
  const rawEvDemandKW = evCount * 7.2;
  let allocatedEvKW = rawEvDemandKW;
  if (mode === 'ai') {
    const availableSolarHeadroom = Math.max(40, predictedSolarKW - predictedFacilityKW + 160);
    allocatedEvKW = Math.min(rawEvDemandKW * 1.35, availableSolarHeadroom);
  }

  res.json({
    timestamp: new Date().toISOString(),
    predictedSolarKW,
    predictedFacilityKW,
    allocatedEvKW: Math.round(allocatedEvKW),
    netGridKW: Math.round(predictedSolarKW - (predictedFacilityKW + allocatedEvKW))
  });
});

app.listen(4000, () => console.log('SolarPark Backend running on :4000'));
